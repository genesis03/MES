# 1. 상태 확인
git status

# 2. 수정된 모든 파일 스테이징
git add .

# 3. 커밋 메시지 작성
git commit -m "Design: 라벨 폰트 및 바코드 여백 조정"

# 4. 원격 저장소(GitHub)로 업로드
git push origin main

# 원격 저장소 최신 코드를 로컬로 병합
git pull origin main

# 한줄 명령어(app.py)
git add app.py; git commit -m "app.py 경량화 및 엑셀 정렬 기능 반영"; git push

# 로컬 파일은 남겨두고 Git 추적(GitHub)에서만 제거할 때
git rm --cached "label_manual - 복사본.html"
git commit -m "Remove: 백업 복사본 파일 추적 제외"
git push origin main

# 로컬 복사본 파일을 완전히 삭제할 때 (PowerShell)
Remove-Item "label_manual - 복사본.html"

# 현재 디렉토리의 모든 수정된 파일을 최신 커밋 상태로 복원
git checkout HEAD -- .

# 특정 파일만 복원할 때
git checkout HEAD -- static/css/label_common.css
